"use client";
import { useCallback, useEffect, useMemo, useState } from "react";
import { toast } from "sonner";
import SiteNav from "@/components/SiteNav";
import OtpLogin from "@/components/OtpLogin";
import { useSession } from "@/components/useSession";
import { supabase } from "@/lib/supabase";

type Row = { id: string; full_name: string; email: string; phone: string | null; city: string | null; college: string | null; year_of_study: string | null; branch: string | null; builder_id: string; consent: boolean | null; status: string; submitted_at: string };
type Mod = { user_id: string; email: string; added_at: string };

function group(rows: Row[], key: "college" | "city" | "year_of_study" | "branch"): [string, number][] {
  const m = new Map<string, number>();
  rows.forEach((r) => {
    const k = (r[key] ?? "").toString().trim() || "Not given";
    m.set(k, (m.get(k) ?? 0) + 1);
  });
  return [...m.entries()].sort((a, b) => b[1] - a[1]).slice(0, 8);
}

function perDay(rows: Row[]): [string, number][] {
  const m = new Map<string, number>();
  rows.forEach((r) => {
    const d = new Date(r.submitted_at).toLocaleDateString("en-CA");
    m.set(d, (m.get(d) ?? 0) + 1);
  });
  return [...m.entries()].sort((a, b) => a[0].localeCompare(b[0])).slice(-14);
}

function Bars({ title, data }: { title: string; data: [string, number][] }) {
  const max = Math.max(1, ...data.map(([, n]) => n));
  return (
    <div className="rounded-xl border border-line bg-panel p-4">
      <h3 className="mb-3 text-sm font-semibold">{title}</h3>
      {data.length === 0 ? <p className="text-sm text-slate-400">No data yet.</p> : (
        <ul className="space-y-2">
          {data.map(([k, n]) => (
            <li key={k} className="text-sm">
              <div className="flex justify-between"><span className="truncate pr-2">{k}</span><span className="text-slate-400">{n}</span></div>
              <div className="mt-1 h-2 rounded bg-white/10"><div className="h-2 rounded bg-accent" style={{ width: `${(n / max) * 100}%` }} /></div>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

export default function AdminPage() {
  const { session, loading } = useSession();
  const [role, setRole] = useState<string | null | undefined>(undefined);
  const [rows, setRows] = useState<Row[]>([]);
  const [mods, setMods] = useState<Mod[]>([]);
  const [modEmail, setModEmail] = useState("");
  const [busy, setBusy] = useState(false);

  const loadMods = useCallback(async () => {
    const { data, error } = await supabase.rpc("list_moderators");
    if (error) return toast.error(error.message);
    setMods((data ?? []) as Mod[]);
  }, []);

  useEffect(() => {
    if (!session) return;
    supabase.rpc("my_role").then(async ({ data }) => {
      setRole((data as string) ?? null);
      if (data === "admin") {
        const res = await supabase.from("applications").select("*").order("submitted_at", { ascending: false });
        if (res.error) return toast.error(res.error.message);
        setRows(res.data as Row[]);
        loadMods();
      }
    });
  }, [session, loadMods]);

  async function addMod(e: React.FormEvent) {
    e.preventDefault();
    const email = modEmail.trim();
    if (!email) return;
    setBusy(true);
    const { error } = await supabase.rpc("add_moderator", { p_email: email });
    setBusy(false);
    if (error) return toast.error(error.message);
    toast.success("Moderator added");
    setModEmail("");
    loadMods();
  }

  async function removeMod(id: string, email: string) {
    if (!confirm(`Remove ${email} as a moderator?`)) return;
    const { error } = await supabase.rpc("remove_moderator", { p_user_id: id });
    if (error) return toast.error(error.message);
    toast.success("Moderator removed");
    loadMods();
  }

  const count = (s: string) => rows.filter((r) => r.status === s).length;
  const byCollege = useMemo(() => group(rows, "college"), [rows]);
  const byCity = useMemo(() => group(rows, "city"), [rows]);
  const byYear = useMemo(() => group(rows, "year_of_study"), [rows]);
  const byBranch = useMemo(() => group(rows, "branch"), [rows]);
  const days = useMemo(() => perDay(rows), [rows]);

  function exportCsv() {
    const esc = (v: unknown) => `"${String(v ?? "").replace(/"/g, '""')}"`;
    const head = ["Name", "Email", "Phone", "City", "College", "Year", "Branch", "Builder ID", "Consent", "Status", "Submitted"];
    const lines = rows.map((r) => [r.full_name, r.email, r.phone, r.city, r.college, r.year_of_study, r.branch, r.builder_id, r.consent ? "Yes" : "No", r.status, r.submitted_at].map(esc).join(","));
    const blob = new Blob([[head.join(","), ...lines].join("\n")], { type: "text/csv" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob); a.download = "applications.csv"; a.click();
  }

  return (
    <>
      <SiteNav solid />
      <main className="mx-auto max-w-6xl px-5 py-10">
        <div className="flex flex-wrap items-center justify-between gap-3">
          {role === "admin" ? <h1 className="text-2xl font-bold">Admin overview</h1> : <span />}
          {role === "admin" && <button onClick={exportCsv} className="rounded-lg border border-line px-4 py-2 text-sm hover:bg-white/5">Export CSV</button>}
        </div>
        {loading ? <p className="mt-6 text-slate-400">Loading...</p>
          : !session ? <div className="mt-6 max-w-sm rounded-2xl border border-line bg-panel p-6"><OtpLogin title="Sign in" onDone={() => {}} /></div>
          : role === undefined ? <p className="mt-6 text-slate-400">Checking access...</p>
          : role !== "admin" ? <p className="mt-6 text-slate-400">404 â€“ This page could not be found.</p>
          : (
            <>
              <p className="mt-2 text-sm text-slate-400">Applications are view-only here. Moderators review and decide them.</p>
              <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
                {[["Total registrations", rows.length], ["Pending", count("pending")], ["Accepted", count("accepted")], ["Rejected", count("rejected")]].map(([k, v]) => (
                  <div key={k as string} className="rounded-xl border border-line bg-panel p-4"><p className="text-xs text-slate-400">{k}</p><p className="text-2xl font-bold">{v}</p></div>
                ))}
              </div>

              <section className="mt-6 rounded-xl border border-line bg-panel p-4">
                <h2 className="text-sm font-semibold">Moderators</h2>
                <p className="mt-1 text-xs text-slate-400">The person must log in once on the registration page before you can add them.</p>
                <form onSubmit={addMod} className="mt-3 flex flex-wrap gap-3">
                  <input type="email" required className="min-w-56 flex-1 rounded-lg border border-line bg-ink px-3 py-2 text-sm" placeholder="moderator@email.com" value={modEmail} onChange={(e) => setModEmail(e.target.value)} />
                  <button disabled={busy} className="rounded-lg bg-accent px-4 py-2 text-sm font-semibold text-white disabled:opacity-50">{busy ? "Adding..." : "Add moderator"}</button>
                </form>
                <ul className="mt-4 divide-y divide-line text-sm">
                  {mods.map((m) => (
                    <li key={m.user_id} className="flex items-center justify-between py-2">
                      <span>{m.email}</span>
                      <button onClick={() => removeMod(m.user_id, m.email)} className="rounded border border-red-500/50 px-3 py-1 text-xs text-red-300 hover:bg-red-500/10">Remove</button>
                    </li>
                  ))}
                  {mods.length === 0 && <li className="py-2 text-slate-400">No moderators yet.</li>}

              <div className="mt-6 grid gap-4 md:grid-cols-2">

export default function RegisterPage() {
  const router = useRouter();
  const { session, loading } = useSession();
  const [open, setOpen] = useState(true);
  const [form, setForm] = useState({
    full_name: "", phone: "", city: "", college: "",
    year_of_study: "", branch: "", builder_id: "", consent: false,
  });
  const [busy, setBusy] = useState(false);

  // If already applied, go straight to the status page. Also check if applications are open.
  useEffect(() => {
    if (!session) return;
    supabase.from("applications").select("id").eq("user_id", session.user.id).maybeSingle().then(({ data }) => {
      if (data) router.replace("/status");
    });
  }, [session, router]);
  useEffect(() => {
    supabase.rpc("applications_open").then(({ data }) => { if (data === false) setOpen(false); });
  }, []);

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    if (!session) return;
    const phone = form.phone.replace(/\D/g, "").slice(-10);
    if (phone.length !== 10) return toast.error("Enter a valid 10-digit phone number.");
    if (!form.consent) return toast.error("Please accept the consent checkbox.");
    setBusy(true);
    const { error } = await supabase.from("applications").insert({
      user_id: session.user.id,
      email: session.user.email,
      full_name: form.full_name.trim(),
      phone,
      city: form.city.trim(),
      college: form.college.trim() || null,
      year_of_study: form.year_of_study,
      branch: form.branch.trim(),
      builder_id: form.builder_id.trim(),
      consent: form.consent,
    });
    setBusy(false);
    if (error) return toast.error(error.code === "23505" ? "You have already applied." : error.message);
    toast.success("Registration submitted!");
    router.push("/status");
  }

  return (

      <main className="mx-auto grid max-w-6xl gap-10 px-5 py-12 lg:grid-cols-5">

          <dl className="space-y-3 rounded-xl border border-line bg-panel p-5 text-sm">
            <div><dt className="text-slate-400">Date</dt><dd className="font-semibold">{EVENT.dateLabel} Â· {EVENT.timeLabel}</dd></div>
            <div><dt className="text-slate-400">Venue</dt><dd className="font-semibold">{EVENT.venue}, {EVENT.city}</dd></div>
            <div><dt className="text-slate-400">Entry</dt><dd className="font-semibold">Free Â· limited seats</dd></div>
          </dl>
          <ol className="space-y-3 text-sm">
            {STEPS.map((s) => (
              <li key={s.n} className="flex gap-3">
                <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-full bg-accent text-xs font-bold text-white">{s.n}</span>

===== components\Countdown.tsx =====

"use client";
import { useEffect, useState } from "react";

export default function Countdown({ iso }: { iso: string }) {
  const [left, setLeft] = useState<number | null>(null);
  useEffect(() => {
    if (!iso) return;
    const t = new Date(iso).getTime();
    const f = () => setLeft(t - Date.now());
    f();
    const id = setInterval(f, 30000);
    return () => clearInterval(id);
  }, [iso]);
  if (!iso || left === null || left <= 0) return null;
  const d = Math.floor(left / 864e5), h = Math.floor((left % 864e5) / 36e5), m = Math.floor((left % 36e5) / 6e4);
  return (
    <div className="mt-6 flex gap-3">
      {[["Days", d], ["Hours", h], ["Minutes", m]].map(([l, v]) => (
        <div key={l as string} className="w-20 rounded-xl bg-white/10 py-3 text-center backdrop-blur">
          <p className="text-2xl font-semibold">{v}</p><p className="text-[11px] uppercase tracking-wider text-slate-300">{l}</p>
        </div>
      ))}
    </div>
  );
}


===== components\CountUp.tsx =====

"use client";
import { useEffect, useRef, useState } from "react";

export default function CountUp({ to, suffix = "" }: { to: number; suffix?: string }) {
  const ref = useRef<HTMLSpanElement>(null);
  const [n, setN] = useState(to);
  useEffect(() => {
    const el = ref.current;
    if (!el || window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    setN(0);
    const io = new IntersectionObserver(([e]) => {
      if (!e.isIntersecting) return;
      io.disconnect();
      const t0 = performance.now();
      const tick = (t: number) => {
        const p = Math.min((t - t0) / 1400, 1);
        setN(Math.round(to * (1 - Math.pow(1 - p, 3))));
        if (p < 1) requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    }, { threshold: 0.4 });
    io.observe(el);
    return () => io.disconnect();
  }, [to]);
  return <span ref={ref}>{n}{suffix}</span>;
}

"use client";
import { useState } from "react";
import { toast } from "sonner";
import { supabase } from "@/lib/supabase";

const input = "w-full rounded-lg border border-line bg-ink px-4 py-3 text-sm outline-none focus:border-accent";
const btn = "w-full rounded-lg bg-accent px-4 py-3 text-sm font-semibold text-white hover:brightness-110 disabled:opacity-50";

export default function OtpLogin({ onDone, title = "Verify your email" }: { onDone: () => void; title?: string }) {
  const [email, setEmail] = useState("");
  const [code, setCode] = useState("");
  const [sent, setSent] = useState(false);
  const [busy, setBusy] = useState(false);

  async function send(e: React.FormEvent) {
    e.preventDefault();
    setBusy(true);
    const { error } = await supabase.auth.signInWithOtp({ email: email.trim().toLowerCase() });
    setBusy(false);
    if (error) return toast.error(error.message);
    setSent(true);
    toast.success("Code sent. Check your inbox (and spam).");
  }

  async function verify(e: React.FormEvent) {
    e.preventDefault();
    setBusy(true);
    const { error } = await supabase.auth.verifyOtp({ email: email.trim().toLowerCase(), token: code.trim(), type: "email" });
    setBusy(false);
    if (error) return toast.error(error.message);
    onDone();
  }

  return (
    <div className="space-y-4">
      <h2 className="text-lg font-semibold">{title}</h2>
      {!sent ? (
        <form onSubmit={send} className="space-y-3">
          <input className={input} type="email" required placeholder="you@example.com" value={email} onChange={(e) => setEmail(e.target.value)} />
          <button className={btn} disabled={busy}>{busy ? "Sending..." : "Send code"}</button>
        </form>
      ) : (
        <form onSubmit={verify} className="space-y-3">
          <p className="text-sm text-slate-400">Enter the code sent to {email}</p>
          <input className={`${input} tracking-[0.4em]`} inputMode="numeric" required placeholder="Code" value={code} onChange={(e) => setCode(e.target.value)} />
          <button className={btn} disabled={busy}>{busy ? "Verifying..." : "Verify"}</button>
          <button type="button" className="w-full text-xs text-slate-400 underline" onClick={() => setSent(false)}>Use a different email</button>
        </form>
      )}
    </div>
  );
}

"use client";
import { useEffect, useRef, useState } from "react";

export default function Photo({ src, alt, className = "", hideOnError = false }: { src?: string; alt: string; className?: string; hideOnError?: boolean }) {
  const [bad, setBad] = useState(false);
  const ref = useRef<HTMLImageElement>(null);
  // The image may already have failed before React hydrated, so check once on mount.
  useEffect(() => {
    const el = ref.current;
    if (el && el.complete && el.naturalWidth === 0) setBad(true);
  }, []);
  if (!src || bad) {
    if (hideOnError) return null;
    return <div role="img" aria-label={alt} className={`bg-gradient-to-br from-indigo-950 via-purple-900 to-fuchsia-900 ${className}`} />;
  }
  // eslint-disable-next-line @next/next/no-img-element
  return <img ref={ref} src={src} alt={alt} className={`object-cover ${className}`} onError={() => setBad(true)} />;
}

"use client";
import { useEffect, useRef, useState } from "react";

export default function Reveal({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
  const ref = useRef<HTMLDivElement>(null);
  const [ready, setReady] = useState(false);
  const [on, setOn] = useState(false);
  useEffect(() => {
    setReady(true);
    const el = ref.current;
    if (!el) return;
    const io = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setOn(true); io.disconnect(); } }, { threshold: 0.1 });
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return (
    <div ref={ref} style={{ transitionDelay: `${delay}ms` }}
      className={`${ready && !on ? "translate-y-6 opacity-0" : "translate-y-0 opacity-100"} transition duration-700 ease-out motion-reduce:transition-none ${className}`}>
      {children}
    </div>
  );
}

"use client";
import { useEffect, useState } from "react";
import type { Session } from "@supabase/supabase-js";
import { supabase } from "@/lib/supabase";

export function useSession() {
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => { setSession(data.session); setLoading(false); });
    const { data: sub } = supabase.auth.onAuthStateChange((_e, s) => setSession(s));
    return () => sub.subscription.unsubscribe();
  }, []);
  return { session, loading };
}

-- Run this whole file once in Supabase: SQL Editor > New query > Run.

create table if not exists public.user_roles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  role text not null check (role in ('admin','moderator')),
  created_at timestamptz not null default now()
);

create table if not exists public.event_settings (
  id int primary key default 1 check (id = 1),     -- single row
  applications_open boolean not null default true,
  whatsapp_group_url text
);
insert into public.event_settings (id) values (1) on conflict do nothing;

create table if not exists public.applications (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null unique references auth.users(id) on delete cascade,
  email text not null,
  full_name text not null,
  builder_id text not null,
  college text,
  status text not null default 'pending' check (status in ('pending','accepted','rejected')),
  submitted_at timestamptz not null default now(),
  reviewed_by uuid references auth.users(id),
  reviewed_at timestamptz
);

create table if not exists public.application_decisions (
  id uuid primary key default gen_random_uuid(),
  application_id uuid not null references public.applications(id) on delete cascade,
  reviewer_id uuid not null references auth.users(id),
  decision text not null,
  review_note text,
  decided_at timestamptz not null default now()
);

-- Helpers -------------------------------------------------------------
create or replace function public.is_staff() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.user_roles where user_id = auth.uid());
$$;

create or replace function public.applications_open() returns boolean
language sql stable security definer set search_path = public as $$
  select applications_open from public.event_settings where id = 1;
$$;

-- WhatsApp link is returned ONLY to the user whose application is accepted.
create or replace function public.get_whatsapp_link() returns text
language sql stable security definer set search_path = public as $$
  select s.whatsapp_group_url
  from public.event_settings s
  where exists (select 1 from public.applications a where a.user_id = auth.uid() and a.status = 'accepted');
$$;

create or replace function public.review_application(app_id uuid, new_status text, note text default null)
returns void language plpgsql security definer set search_path = public as $$
begin
  if not public.is_staff() then raise exception 'not allowed'; end if;
  if new_status not in ('accepted','rejected','pending') then raise exception 'bad status'; end if;
  update public.applications set status = new_status, reviewed_by = auth.uid(), reviewed_at = now() where id = app_id;
  insert into public.application_decisions (application_id, reviewer_id, decision, review_note)
  values (app_id, auth.uid(), new_status, note);
end $$;

-- Row level security ----------------------------------------------------
alter table public.user_roles enable row level security;
alter table public.event_settings enable row level security;
alter table public.applications enable row level security;
alter table public.application_decisions enable row level security;

-- applicants: read their own row, insert their own row as pending only. No updates.
create policy "own application read" on public.applications for select using (user_id = auth.uid());
create policy "staff read all" on public.applications for select using (public.is_staff());
create policy "own application insert" on public.applications for insert
  with check (user_id = auth.uid() and status = 'pending' and public.applications_open());

create policy "staff read decisions" on public.application_decisions for select using (public.is_staff());
create policy "own role read" on public.user_roles for select using (user_id = auth.uid());
-- event_settings has no policies: clients can't read the WhatsApp link directly (only via get_whatsapp_link()).

grant execute on function public.is_staff() to authenticated, anon;
grant execute on function public.applications_open() to authenticated, anon;
grant execute on function public.get_whatsapp_link() to authenticated;
grant execute on function public.review_application(uuid, text, text) to authenticated;
